/**
 * Day bucketing for the notification list headers ("Dzisiaj" / "Wczoraj" / a
 * date). Pure and defensive: the feed is written by several producers, so one
 * row with a broken `createdAt` must not take the whole screen down — it lands
 * in its own trailing bucket instead.
 */
}

/**
 * "today" / "yesterday" / `null` (render the date instead). Compares calendar
 * days in LOCAL time, not elapsed hours — 23:59 yesterday is "Wczoraj" even
 * when it is two minutes old.
 */
export function relativeDayKind(key, now = Date.now()) {
  if (typeof key !== "string" || key === "unknown") return null;
  const today = new Date(now);
  if (Number.isNaN(today.getTime())) return null;
  if (key === dayKeyOf(today)) return "today";
  // Calendar arithmetic, not `now - 24h`: the day a DST shift lands on is 23 or
  // 25 hours long and the subtraction would name the wrong date.
  const yesterday = new Date(today.getFullYear(), today.getMonth(), today.getDate() - 1);
  return key === dayKeyOf(yesterday) ? "yesterday" : null;
}

## Playmore — src/modules/notificationCenter/components/NotificationDayGroup.vue (heading)
const heading = computed(() => {
  const kind = relativeDayKind(props.dayKey);
  if (kind) return t(`notificationCenter.day.${kind}`);
  if (!props.date) return t("notificationCenter.day.unknown");
  // Locale from useI18n, never a hardcoded pl-PL — the year only when it is not
  // the current one, so the common case stays short ("12 lipca").
  const sameYear = props.date.getFullYear() === new Date().getFullYear();
  return props.date.toLocaleDateString(locale.value, {
    day: "numeric",
    month: "long",
    ...(sameYear ? {} : { year: "numeric" }),
  });
});
</script>
